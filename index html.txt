<script>
const CSV_URL = "PASTE_CSV_URL_DI_SINI";

function parseCSV(text){
  const rows = [];
  let row = [], cur = "", inQuotes = false;
  for (let i=0;i<text.length;i++){
    const c = text[i], n = text[i+1];
    if (c === '"' && inQuotes && n === '"'){ cur += '"'; i++; continue; }
    if (c === '"'){ inQuotes = !inQuotes; continue; }
    if (c === ',' && !inQuotes){ row.push(cur); cur=""; continue; }
    if (c === '\n' && !inQuotes){ row.push(cur); rows.push(row); row=[]; cur=""; continue; }
    cur += c;
  }
  row.push(cur); rows.push(row);
  return rows.filter(r => r.some(x => (x||"").trim() !== ""));
}

function rowToBook(headers, r){
  const get = (name) => r[headers.indexOf(name)] || "";
  return {
    id: get("ID"),
    title: get("Judul Buku"),
    subtitle: get("Subjudul"),
    desc: get("Deskripsi Singkat"),
    kategori: get("Kategori"),
    target: get("Target Pembaca"),
    tahun: get("Tahun Terbit"),
    cover: get("Cover (URL)"),
    shopee: get("Link Shopee"),
    tokopedia: get("Link Tokopedia"),
    penerbit: get("Link Website Penerbit"),
  };
}

let BOOKS = [];

function render(filter=""){
  const f = filter.trim().toLowerCase();
  const grid = document.getElementById("grid");
  grid.innerHTML = "";

  BOOKS.filter(b => {
    const hay = [b.title,b.subtitle,b.desc,b.kategori,b.target,b.tahun].join(" ").toLowerCase();
    return hay.includes(f);
  }).forEach(b => {
    const card = document.createElement("div");
    card.className = "card";
    card.innerHTML = `
      <div class="cover">${b.cover ? `<img src="${b.cover}" alt="Cover ${b.title}">` : ""}</div>
      <div class="content">
        <p class="title">${b.title || ""}</p>
        <p class="meta">${[b.subtitle, b.tahun].filter(Boolean).join(" • ")}</p>
        <p class="desc">${b.desc || ""}</p>
        <div class="btns">
          ${b.shopee ? `<a class="btn primary" href="${b.shopee}" target="_blank" rel="noopener">Beli Shopee</a>` : ""}
          ${b.tokopedia ? `<a class="btn" href="${b.tokopedia}" target="_blank" rel="noopener">Beli Tokopedia</a>` : ""}
          ${b.penerbit ? `<a class="btn" href="${b.penerbit}" target="_blank" rel="noopener">Detail</a>` : ""}
        </div>
      </div>
    `;
    grid.appendChild(card);
  });
}

async function init(){
  const res = await fetch(CSV_URL);
  const text = await res.text();
  const rows = parseCSV(text);
  const headers = rows[0].map(h => h.trim());
  BOOKS = rows.slice(1).map(r => rowToBook(headers, r));
  render();
}

document.getElementById("q").addEventListener("input", e => render(e.target.value));
document.getElementById("y").textContent = new Date().getFullYear();
init();
